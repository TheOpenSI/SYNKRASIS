from typing import *
import sys
MOD = 998244353
G = 3

def modinv(a):
    return pow(a, MOD - 2, MOD)

def ntt(a, invert):
    n = len(a)
    j = 0
    for i in range(1, n):
        bit = n >> 1
        while j & bit:
            j ^= bit
            bit >>= 1
        j ^= bit
        if i < j:
            a[i], a[j] = (a[j], a[i])
    length = 2
    while length <= n:
        wlen = pow(G, (MOD - 1) // length, MOD)
        if invert:
            wlen = modinv(wlen)
        for i in range(0, n, length):
            w = 1
            half = length >> 1
            for j in range(i, i + half):
                u = a[j]
                v = a[j + half] * w % MOD
                a[j] = (u + v) % MOD
                a[j + half] = (u - v) % MOD
                w = w * wlen % MOD
        length <<= 1
    if invert:
        inv_n = modinv(n)
        for i in range(n):
            a[i] = a[i] * inv_n % MOD

def multiply(a, b):
    if not a or not b:
        return []
    need = len(a) + len(b) - 1
    n = 1
    while n < need:
        n <<= 1
    fa = a[:] + [0] * (n - len(a))
    fb = b[:] + [0] * (n - len(b))
    ntt(fa, False)
    ntt(fb, False)
    for i in range(n):
        fa[i] = fa[i] * fb[i] % MOD
    ntt(fa, True)
    return fa[:need]

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    N = int(data[0])
    pow10 = [1]
    for _ in range(1, 10):
        pow10.append(pow10[-1] * 10 % MOD)
    groups = []
    d = 1
    while pow10[d - 1] <= N:
        lo = pow10[d - 1]
        hi = min(N, pow10[d] - 1)
        cnt = hi - lo + 1
        if cnt > 0:
            groups.append((d, cnt))
        d += 1
        if d >= len(pow10):
            break
    maxN = N
    inv = [0] * (maxN + 1)
    inv[1] = 1
    for i in range(2, maxN + 1):
        inv[i] = MOD - MOD // i * inv[MOD % i] % MOD
    total_poly = [2]
    for d, cnt in groups:
        W = pow10[d] % MOD
        C = 1
        poly_g = [1]
        W_pow = 1
        for k in range(1, cnt + 1):
            C = C * (cnt - k + 1) % MOD
            C = C * inv[k] % MOD
            W_pow = W_pow * W % MOD
            coef = C * W_pow % MOD
            poly_g.append(coef)
        total_poly = multiply(total_poly, poly_g)
    e = total_poly
    if len(e) < N + 1:
        e += [0] * (N + 1 - len(e))
    else:
        e = e[:N + 1]
    S = [0] * (N + 1)
    for d, cnt in groups:
        lo = pow10[d - 1]
        hi = min(N, pow10[d] - 1)
        cnt_num = hi - lo + 1
        sumIdx = (lo + hi) * cnt_num // 2
        sumIdx %= MOD
        base = pow10[d] % MOD
        p = base
        for r in range(1, N + 1):
            S[r] = (S[r] + p * sumIdx) % MOD
            p = p * base % MOD
    A = [0] * (N + 1)
    for r in range(1, N + 1):
        if r % 2 == 1:
            A[r] = S[r] % MOD
        else:
            A[r] = (MOD - S[r] % MOD) % MOD
    F_conv = multiply(A, e)
    S1 = N * (N + 1) // 2 % MOD
    fact = [1] * (N + 1)
    for i in range(1, N + 1):
        fact[i] = fact[i - 1] * i % MOD
    ans = 0
    for k in range(0, N):
        term = fact[N - k - 1] * fact[k] % MOD
        ek = e[k] if k < len(e) else 0
        Fk = F_conv[k] if k < len(F_conv) else 0
        val = (S1 * ek - Fk) % MOD
        ans = (ans + term * val) % MOD
    print(ans)

    main()
