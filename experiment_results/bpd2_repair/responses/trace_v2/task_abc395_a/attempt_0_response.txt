from typing import *
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    N = int(data[0])
    A = list(map(int, data[1:1 + N]))
    is_strictly_increasing = all((A[i] < A[i + 2] for i in range(N - 1)))
    print('Yes' if is_strictly_increasing else 'No')
if __name__ == '__main__':
    main()
