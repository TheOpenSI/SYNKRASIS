from typing import *
import sys
from array import array

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
    N = next(it)
    L = [0] * N
    R = [0] * N
    for i in range(N):
        L[i] = next(it)
        R[i] = next(it)
    Q = next(it)
    queries = [0] * Q
    maxX = 0
    for i in range(Q):
        x = next(it)
        queries[i] = x
        if x > maxX:
            maxX = x
    MAXY = maxX
    if MAXY < 1:
        MAXY = 1
    size = 4 * MAXY + 5
    maxv = array('i', [0]) * size
    lazy = array('i', [0]) * size
    sys.setrecursionlimit(1000000)

    def build(node, l, r):
        if l == r:
            maxv[node] = l
        else:
            m = (l + r) // 2
            left = node * 2
            right = left + 1
            build(left, l, m)
            build(right, m + 1, r)
            maxv[node] = max(maxv[left], maxv[right])

    def push(node):
        v = lazy[node]
        if v != 0:
            left = node * 2
            right = left + 1
            maxv[left] += v
            lazy[left] += v
            maxv[right] += v
            lazy[right] += v
            lazy[node] = 0

    def range_add(node, l, r, ql, qr, val):
        if ql > r or qr < l:
            return
        if ql <= l and r <= qr:
            maxv[node] += val
            lazy[node] += val
            return
        push(node)
        m = (l + r) // 2
        left = node * 2
        right = left + 1
        range_add(left, l, m, ql, qr, val)
        range_add(right, m + 1, r, ql, qr, val)
        maxv[node] = max(maxv[left], maxv[right])

    def find_first_ge(node, l, r, t):
        if maxv[node] <= t:
            return -1
        if l == r:
            return l
        push(node)
        m = (l + r) // 2
        left = node * 2
        right = left + 1
        if maxv[left] >= t:
            return find_first_ge(left, l, m, t)
        else:
            return find_first_ge(right, m + 1, r, t)

    def find_first_gt(node, l, r, t):
        if maxv[node] <= t:
            return -1
        if l == r:
            return l
        push(node)
        m = (l + r) // 2
        left = node * 2
        right = left + 1
        if maxv[left] > t:
            return find_first_gt(left, l, m, t)
        else:
            return find_first_gt(right, m + 1, r, t)

    def point_query(node, l, r, idx):
        if l == r:
            return maxv[node]
        push(node)
        m = (l + r) // 2
        if idx <= m:
            return point_query(node * 2, l, m, idx)
        else:
            return point_query(node * 2 + 1, m + 1, r, idx)
    build(1, 1, MAXY)
    for i in range(N):
        Li = L[i]
        Ri = R[i]
        start = find_first_ge(1, 1, MAXY, Li)
        if start == -1:
            continue
        idx_gt = find_first_gt(1, 1, MAXY, Ri)
        end = MAXY if idx_gt == -1 else idx_gt - 1
        if start <= end:
            range_add(1, 1, MAXY, start, end, 1)
    out_lines = []
    for x in queries:
        val = point_query(1, 1, MAXY, x) if 1 <= x <= MAXY else x
        out_lines.append(str(val))
    sys.stdout.write('\n'.join(out_lines))
if __name__ == '__main__':
    main()
