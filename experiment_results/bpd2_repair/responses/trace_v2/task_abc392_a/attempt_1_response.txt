- The correct approach is to check all possible choices of B3 among the three numbers. For each choice, check if the product of the other two numbers equals that chosen B3.
- If any such choice satisfies B1*B2 = B3, the answer is Yes; otherwise No.
- This handles duplicates and all permutations safely with a simple O(1) check over 3 cases.

import sys

def main():
    data = sys.stdin.read().strip().split()
    if len(data) < 3:
        print("No")
        return
    nums = list(map(int, data[:3]))
    possible = False
    for i in range(3):
        prod = nums[i]
        others = [nums[j] for j in range(3) if j != i]
        if others[0] * others[1] == prod:
            possible = True
            break
    print("Yes" if possible else "No")

if __name__ == "__main__":
    main()
