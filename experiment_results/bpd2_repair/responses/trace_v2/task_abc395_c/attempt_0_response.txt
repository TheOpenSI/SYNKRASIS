from typing import *
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        print(-1)
        return
    n = data[1]
    a = data[1:1 + n]
    last_pos = {}
    ans = None
    for i, v in enumerate(a):
        if v in last_pos:
            length = i - last_pos[v] + 1
            if ans is None or length < ans:
                ans = length
        last_pos[v] = i
    print(-1 if ans is None else ans)
if __name__ == '__main__':
    main()
