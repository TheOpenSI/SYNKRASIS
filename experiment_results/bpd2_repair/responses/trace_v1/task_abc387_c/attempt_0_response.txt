from typing import *
import sys

def count_snake_upto(N: int) -> int:
    if N < 10:
        return 0
    s = str(N)
    L = len(s)
    total = 0
    for k in range(2, L):
        p = k - 1
        total += sum((a ** p for a in range(1, 10)))
    first = int(s[0])
    for a in range(1, first):
        total += a ** (L - 1)
    a_lim = first - 1
    B = [int(ch) for ch in s[1:]]
    dp_tight = 1
    dp_loose = 0
    for bi in B:
        new_tight = 0
        new_loose = 0
        if dp_tight:
            max_less = min(a_lim, bi - 1)
            if max_less >= 0:
                new_loose -= dp_tight * (max_less + 1)
            if bi <= a_lim:
                new_tight += dp_tight
        if dp_loose:
            new_loose += dp_loose * (a_lim + 1)
        dp_tight, dp_loose = (new_tight, new_loose)
    total += dp_tight + dp_loose
    return total

def main():
    data = sys.stdin.read().strip().split()
    L = int(data[0])
    R = int(data[1])
    ans = count_snake_upto(R) - count_snake_upto(L - 1)
    print(ans)
if __name__ == '__main__':
    main()
