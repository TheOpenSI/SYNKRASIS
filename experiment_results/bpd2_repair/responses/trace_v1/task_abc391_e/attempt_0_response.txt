from typing import *
from array import array
import sys

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    A = b''.join(data[1:]).decode()
    L = 3 ** N
    if len(A) > L:
        A = A[:L]
    dp0 = array('i', [0]) * L
    dp1 = array('i', [0]) * L
    for i in range(L):
        if A[i] == '0':
            dp0[i] = 0
            dp1[i] = 2
        else:
            dp0[i] = 1
            dp1[i] = 0
    size = L
    while size > 1:
        new_size = size // 3
        ndp0 = array('i', [0]) * new_size
        ndp1 = array('i', [0]) * new_size
        for j in range(new_size):
            i0 = 3 * j
            a0 = dp0[i0]
            a1 = dp1[i0]
            b0 = dp0[i0 + 1]
            b1 = dp1[i0 + 1]
            c0 = dp0[i0 + 2]
            c1 = dp1[i0 + 2]
            v0 = a0 + b0 + c0
            v1 = a0 + b0 + c1
            v2 = a0 + b1 + c0
            v3 = a1 + b0 + c0
            m0 = v0
            if v1 < m0:
                m0 = v1
            if v2 < m0:
                m0 = v2
            if v3 < m0:
                m0 = v3
            t1 = a1 + b1 + c1

            t4 = a0 + b1 + c1
            m1 = t1
            if t2 < m1:
                m1 = t2
            if t3 < m1:
                m1 = t3
            if t4 < m1:
                m1 = t4
            ndp0[j] = m0
            ndp1[j] = m1
        dp0, dp1 = (ndp0, ndp1)
        size = new_size
    root0 = int(dp0[0])
    root1 = int(dp1[0])
    if root0 == 0:
        print(root1)
    else:

if __name__ == '__main__':
