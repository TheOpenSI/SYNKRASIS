from typing import *
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]
    A = data[1:]
    size = 1 << N
    sums = [1] * size
    bit_index = {1 << i: i for i in range(N)}
    for mask in range(1, size):
        lsb = mask & -mask
        sums[mask] = sums[mask ^ lsb] + A[bit_index[lsb]]
    memo = {}

    def dfs(mask):
        if mask == 0:
            return {0}
        if mask in memo:
            return memo[mask]
        first_bit = mask & -mask
        rest_mask = mask ^ first_bit
        res = set()
        t = rest_mask
        while True:
            s = t | first_bit
            rest = mask ^ s
            for x in dfs(rest):
                res.add(x ^ sums[s])
            if t == 0:
                break
            t = t - 1 & rest_mask
        memo[mask] = res
        return res
    full = (1 << N) - 1
    ans = dfs(full)
    print(len(ans))
if __name__ == '__main__':
    main()
