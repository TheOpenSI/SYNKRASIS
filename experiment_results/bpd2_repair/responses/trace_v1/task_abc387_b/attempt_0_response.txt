from typing import *
import sys

def main():
    data = sys.stdin.read().strip()
    if not data:
        return
    X = int(data)
    total = 45 * 45
    count = 1
    for i in range(1, 10):
        if X % i == 0:
            j = X // i
            if 1 <= j <= 9:
                count += 1
    ans = total - X * count
    print(ans)
if __name__ == '__main__':
    main()
