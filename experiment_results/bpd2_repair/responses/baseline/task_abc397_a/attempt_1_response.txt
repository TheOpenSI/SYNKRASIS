### Code
```
import sys
from decimal import Decimal

def main():
    s = sys.stdin.readline().strip()
    if not s:
        return
    x = Decimal(s)
    if x >= Decimal('38.0'):
        print(1)
    elif x >= Decimal('37.5'):
        print(2)
    else:
        print(3)

if __name__ == '__main__':
    main()
```
