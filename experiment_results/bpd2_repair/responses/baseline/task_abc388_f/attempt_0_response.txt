from typing import *
import sys

def main():
    data = sys.stdin.read().strip().split()
    it = iter(data)
    N = int(next(it))
    M = int(next(it))
    A = int(next(it))
    B = int(next(it))
    intervals = []
    for _ in range(M):
        L = int(next(it))
        R = int(next(it))
        intervals.append((L, R))
    prefix_mask = (1 << B - A + 1) - 1
    maskB = (1 << B) - 1
    current = 1
    Mstate = 1 << B - 1

    def process_safe_block(length):
        nonlocal current, Mstate
        if length <= 0:
            return False
        end = current - length
        seen = dict()
        while current < end:
            if Mstate in seen:
                cycle_start_pos = seen[Mstate]
                cycle_len = current - cycle_start_pos
                rem = end - current
                if cycle_len > 0 and rem > 0:
                    if N > end:
                        max_cycles = rem // cycle_len
                    else:
                        max_cycles = (N - current) // cycle_len
                        if max_cycles < 0:
                            max_cycles = 0
                    if max_cycles > 0:
                        if N >= current and N <= end and ((N - current) % cycle_len == 0):
                            max_cycles = max(0, max_cycles - 1)
                        if max_cycles > 0:
                            current += max_cycles * cycle_len
                            if current == N:
                                return False
                            continue
            else:
                seen[Mstate] = current
            Next = 1 if Mstate & prefix_mask != 0 else 0
            Mstate = Mstate >> 1 | Next << B - 1
            current += 1
            if current == N and Next == 1:
                return True
        return False

    def process_bad_block(length):
        nonlocal current, Mstate
        if length <= 0:
            return
        k = min(length, B)
        for _ in range(k):
            current += 1
            Mstate >>= 1
        if length > B:
            current += length - B
            Mstate = 0
    for idx in range(M + 1):
        if idx < M:
            end_safe = intervals[idx][0] - 1
        else:
            end_safe = N
        length_safe = end_safe - current
        if length_safe > 0:
            if process_safe_block(length_safe):
                print('Yes')
                return
        current = end_safe
        if current >= N:
            break
        if idx < M:
            L, R = intervals[idx]
            length_bad = R - L + 1
            process_bad_block(length_bad)
            current = R
            if current >= N:
                break
    print('No')
if __name__ == '__main__':
    main()
