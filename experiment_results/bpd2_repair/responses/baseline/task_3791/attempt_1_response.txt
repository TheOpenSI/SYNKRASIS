from typing import List

class Solution:
    def numOfUnplacedFruits(self, fruits: List[int], baskets: List[int]) -> int:
        n = len(baskets)
        if n == 0:
            return len(fruits)
        INF_NEG = -10 ** 18
        tree = [INF_NEG] * (4 * n)

        def build(node: int, l: int, r: int):
            if l == r:
                tree[node] = baskets[l]
            else:
                mid = (l + r) // 2
                build(node * 2, l, mid)
                build(node * 2 + 1, mid + 1, r)
                left = tree[node * 2]
                right = tree[node * 2 + 1]
                tree[node] = left if left >= right else right

        def update(node: int, l: int, r: int, idx: int, val: int):
            if l == r:
                tree[node] = val
            else:
                mid = (l + r) // 2
                if idx <= mid:
                    update(node * 2, l, mid, idx, val)
                else:
                    update(node * 2 + 1, mid + 1, r, idx, val)
                left = tree[node * 2]
                right = tree[node * 2 + 1]
                tree[node] = left if left >= right else right

        def find_first_ge(node: int, l: int, r: int, x: int) -> int:
            if tree[node] < x:
                return -1
            if l == r:
                return l
            mid = (l + r) // 2
            if tree[node * 2] >= x:
                return find_first_ge(node * 2, l, mid, x)
            else:
                return find_first_ge(node * 2 + 1, mid + 1, r, x)

        build(1, 0, n - 1)
        unplaced = 0
        for f in fruits:
            idx = find_first_ge(1, 0, n - 1, f)
            if idx == -1:
                unplaced += 1
            else:
                update(1, 0, n - 1, idx, INF_NEG)
        return unplaced
