### Step-by-step reasoning
- The problem asks, for each subarray [L, R], the maximum number of disjoint pairs (i, j) with L ≤ i < j ≤ R such that A[i] * 2 ≤ A[j].
- This is a maximum matching problem on a general graph: create a vertex for each position in the subarray, and add an undirected edge between i and j if i < j and A[i] * 2 ≤ A[j]. We need the maximum size of a matching (each mochi used at most once).
- A simple greedy strategy (two-pointer or earliest-possible partner) can fail in this setting, as shown by a small counterexample. Therefore, we need an actual maximum matching algorithm for general graphs.
- The appropriate approach is Edmonds’ blossom algorithm, which finds a maximum matching in general graphs in polynomial time.
- For each query, we:
  - Build the subarray B = A[L:R+1].
  - Build the graph G with n = R-L+1 vertices; add edges (i, j) for i < j if B[i] * 2 ≤ B[j].
  - Run a blossom algorithm to compute the maximum matching size for G.
  - Output that size.

This approach is correct for the required problem since the structure is a general graph, and blossom handles all cases of odd cycles gracefully. If subarray sizes are small in the test suite, this runs quickly enough.

import sys
from collections import deque

def max_matching_general(G):
    """
    Edmonds' Blossom algorithm for maximum matching in general graphs.
    G: adjacency list, G[v] is a list of neighbors of v.
    Returns the size of the maximum matching.
    """
    n = len(G)
    if n == 0:
        return 0

    match = [-1] * n
    base = list(range(n))
    p = [-1] * n
    used = [False] * n
    blossom = [False] * n
    q = deque()

    def lca(a, b):
        used_path = [False] * n
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, u):
        while base[v] != b:
            blossom[base[v]] = True
            blossom[base[match[v]]] = True
            p[v] = u
            u = match[v]
            v = p[match[v]]

    def find_path(root):
        for i in range(n):
            p[i] = -1
            base[i] = i
            used[i] = False
        q.clear()
        q.append(root)
        used[root] = True
        while q:
            v = q.popleft()
            for u in G[v]:
                if base[v] == base[u] or match[v] == u:
                    continue
                if u == root or (match[u] != -1 and p[match[u]] != -1):
                    curbase = lca(v, u)
                    for i in range(n):
                        blossom[i] = False
                    mark_path(v, curbase, u)
                    mark_path(u, curbase, v)
                    for i in range(n):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[u] == -1:
                    p[u] = v
                    if match[u] == -1:
                        # augmenting path found
                        cur = u
                        while cur != -1:
                            prev = p[cur]
                            nxt = match[prev] if prev != -1 else -1
                            match[cur] = prev
                            if prev != -1:
                                match[prev] = cur
                            cur = nxt
                        return True
                    else:
                        w = match[u]
                        if not used[w]:
                            used[w] = True
                            q.append(w)
        return False

    for v in range(n):
        if match[v] == -1:
            find_path(v)

    return sum(1 for i in range(n) if match[i] != -1) // 2

def solve():
    data = sys.stdin.read().strip().split()
    it = iter(data)
    N = int(next(it))
    A = [int(next(it)) for _ in range(N)]
    Q = int(next(it))
    out_lines = []
    for _ in range(Q):
        L = int(next(it)) - 1
        R = int(next(it)) - 1
        m = R - L + 1
        if m < 2:
            out_lines.append("0")
            continue

        # Build subarray graph
        B = A[L:R+1]
        n = m
        G = [[] for _ in range(n)]
        for i in range(n):
            ai = B[i]
            for j in range(i+1, n):
                if ai * 2 <= B[j]:
                    G[i].append(j)
                    G[j].append(i)

        ans = max_matching_general(G)
        out_lines.append(str(ans))

    print("\n".join(out_lines))

if __name__ == "__main__":
    solve()
