import sys

def mat_mul(A, B, mod):
    n = len(A)
    m = len(B[0])
    p = len(B)
    R = [[0] * m for _ in range(n)]
    for i in range(n):
        Ai = A[i]
        Ri = R[i]
        for k in range(p):
            aik = Ai[k]
            if aik == 0:
                continue
            Bk = B[k]
            for j in range(m):
                Ri[j] = (Ri[j] + aik * Bk[j]) % mod
    return R

def mat_pow(A, e, mod):
    n = len(A)
    R = [[0] * n for _ in range(n)]
    for i in range(n):
        R[i][i] = 1 % mod
    X = [row[:] for row in A]
    while e > 0:
        if e & 1:
            R = mat_mul(R, X, mod)
        X = mat_mul(X, X, mod)
        e >>= 1
    return R
