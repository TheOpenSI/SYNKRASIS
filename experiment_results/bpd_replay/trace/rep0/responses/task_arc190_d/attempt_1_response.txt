def mat_mul(A, B, mod):
    n = len(A)
    m = len(B[0])
    p = len(B)
    res = [[0]*m for _ in range(n)]
    for i in range(n):
        Ai = A[i]
        Ri = res[i]
        for k in range(p):
            aik = Ai[k]
            if aik == 0:
                continue
            Bk = B[k]
            for j in range(m):
                Ri[j] = (Ri[j] + aik * Bk[j]) % mod
    return res

def mat_pow(A, e, mod):
