def main():
    data = sys.stdin.read().strip().split()
    it = iter(data)
    N = int(next(it))

    A = [int(next(it)) for _ in range(N)]
    B = [int(next(it)) for _ in range(N)]
    C = [int(next(it)) for _ in range(N)]

    # Baseline: flip each differing bit exactly once, and never touch others.
    # This is NOT guaranteed to be optimal (as test 3 shows), but provides a baseline.
    S0 = 0
    neg = []
    pos = []
    d = 0

    for i in range(N):
        ai = A[i]
        bi = B[i]
        ci = C[i]
        S0 += ai * ci
        di = ai ^ bi
        if di == 1:
            d += 1
            if ai == 1:
                neg.append(ci)  # delta = -ci
            else:
                pos.append(ci)  # delta = +ci

    if d == 0:
        print(0)
        return

    neg.sort(reverse=True)  # largest |ci| first for negative deltas
    pos.sort()              # smallest ci first for positive deltas

    total = d * S0

    # Neg contributions
    for idx, ci in enumerate(neg, start=1):
        weight = d - idx + 1
        total += (-ci) * weight

    # Pos contributions
    base_weight = d - len(neg)
    for t, ci in enumerate(pos, start=1):
        weight = base_weight - t + 1
        total += ci * weight

    print(total)

if __name__ == "__main__":
    main()
