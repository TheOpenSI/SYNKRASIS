import sys

MOD = 998244353

def solve():
    input = sys.stdin.readline
    N_line = input().strip()
    while N_line == "":
        N_line = input().strip()
    N = int(N_line)
    S = input().strip()

    # Quick necessary condition: first vertex must be black, last vertex white
    if not S or S[0] != 'B' or S[-1] != 'W':
        print(0)
        return

    M = 2 * N

    # Compute D[i] = (#W - #B) in prefix S[1..i], with D[0] = 0
    D = [0] * (M + 1)
    w = 0
    for i in range(1, M + 1):
        if S[i - 1] == 'W':
            w += 1
        else:
            w -= 1
        D[i] = w

    # Collect zero positions i in (0, 2N); also include endpoints 0 and 2N
    zeros = [0]
    for i in range(1, M):
        if D[i] == 0:
            zeros.append(i)
    zeros.append(M)  # include 2N

    # Extended zero positions Zext
    Zext = zeros
    L = len(Zext)

    # y_i = Zext[i] / 2 (integers)
    y = [z // 2 for z in Zext]

    # Precompute factorials up to N
    fact = [1] * (N + 1)
    for i in range(1, N + 1):
        fact[i] = (fact[i - 1] * i) % MOD

    # DP over zero-boundaries
    DP = [0] * L
    DP[0] = 1

    for k in range(1, L):
        total = 0
        yk = y[k]
        for j in range(0, k):
            dy = yk - y[j]
            if dy < 0:
                continue
            total = (total + DP[j] * fact[dy]) % MOD
        DP[k] = (-total) % MOD

    # Final answer is -DP[last] mod MOD
    ans = (-DP[L - 1]) % MOD
    print(ans)

if __name__ == "__main__":
    solve()
