from typing import List

class Solution:
    def maxWeight(self, pizzas: List[int]) -> int:
        n = len(pizzas)
        days = n // 4

        # Sort ascending to easily access the largest remaining values via an index
        pizzas.sort()
        R = n - 1  # index of current largest remaining
        total = 0

        for i in range(days):
            if i % 2 == 0:
                # Day 1, 3, ...: gain the largest remaining
                total += pizzas[R]
                R -= 1
            else:
                # Day 2, 4, ...: gain the second largest among the four largest remaining
                total += pizzas[R - 1]
                R -= 4  # remove the four largest remaining

        return total
