from typing import List

class Solution:
    def maxWeight(self, pizzas: List[int]) -> int:
        n = len(pizzas)
        days = n // 4

        # Sort to easily access largest/smallest values
        pizzas.sort()
        R = n - 1  # index of current largest

        ans = 0
        for i in range(days):
            if i % 2 == 0:
                # Day 1, 3, ...: take the largest as Z
                ans += pizzas[R]
                R -= 1
            else:
                # Day 2, 4, ...: take the second largest among the four largest remaining
                ans += pizzas[R - 1]
                R -= 4

        return ans
