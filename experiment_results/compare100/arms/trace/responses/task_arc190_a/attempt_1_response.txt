### Step-by-step reasoning
- The problem asks to minimize the total cost to turn all x_i to 1 using M interval operations, where each operation selects a type (0, 1, or 2) with a given interval [L_i, R_i].
- A single operation of type 1 with an interval covering [1, N] suffices (cost 1). Similarly, two type-2 operations on disjoint or complementary parts can cover all positions (cost 2). There are also cases where a combination of two type-1 operations on two intervals whose union covers [1, N] achieves the goal with cost 2.
- Earlier logic failed on test case 3: intervals [1,3] and [2,5] union to [1,5], so performing type-1 on both intervals (cost 2) is a valid optimal solution, but the previous solution concluded -1.
- To fix this, we add a simple and robust check: if there exist two distinct intervals whose union covers [1, N], we can finish with K=2 using op=1 on both intervals. A practical sufficient condition is:
  - There exists at least one interval with L=1 (call it i1) and at least one interval with R=N (call it j1) such that L(j1) <= R(i1) + 1 and i1 != j1.
  - If so, set op_i1 = 1 and op_j1 = 1 and output K=2.
- Additionally, we keep the previous checks (K=1, disjoint pair for K=2 with op=2, nested pair for K=2 with (2,1)) and the 3-ops/greedy path as a fallback.
- This adjustment fixes test case 3 and preserves existing logic for other cases.

import sys
from collections import namedtuple

Interval = namedtuple('Interval', ['L', 'R', 'idx'])

def solve():
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    N = int(next(it))
    M = int(next(it))
    intervals = []
    for i in range(1, M + 1):
        L = int(next(it)); R = int(next(it))
        intervals.append(Interval(L, R, i))

    ans = [0] * M  # 0 by default

    # 1) K = 1: any interval exactly [1, N]
    for iv in intervals:
        if iv.L == 1 and iv.R == N:
            ans[iv.idx - 1] = 1
            print(1)
            print(' '.join(str(x) for x in ans))
            return

    # 2) Look for disjoint pair to cover all with two op=2
    intervals_by_L = sorted(intervals, key=lambda x: (x.L, x.R))
    maxR = -1
    maxR_idx = -1
    for iv in intervals_by_L:
        if maxR != -1 and iv.L > maxR:
            ans[maxR_idx - 1] = 2
            ans[iv.idx - 1] = 2
            print(2)
            print(' '.join(str(x) for x in ans))
            return
        if iv.R > maxR:
            maxR = iv.R
            maxR_idx = iv.idx

    # 3) Nested pair: A_i ⊆ A_j gives K=2 (i as 2, j as 1)
    maxR_so_far = -1
    maxR_idx_so_far = -1
    for iv in intervals_by_L:
        if maxR_so_far >= iv.R:
            inner_idx = iv.idx
            outer_idx = maxR_idx_so_far
            if inner_idx != outer_idx:
                ans[inner_idx - 1] = 2
                ans[outer_idx - 1] = 1
                print(2)
                print(' '.join(str(x) for x in ans))
                return
        if iv.R > maxR_so_far:
            maxR_so_far = iv.R
            maxR_idx_so_far = iv.idx

    # 4) Quick two-interval union check: if exists i with L=1 and j with R=N, i != j,
    #    and L(j) <= R(i) + 1, we can do K=2 with op=1 on both.
    i1_candidates = [iv for iv in intervals if iv.L == 1]
    if i1_candidates:
        # pick i1 with maximum R among L=1
        i1 = max(i1_candidates, key=lambda x: x.R)
        # candidates for j: those with R=N excluding i1
        j_candidates = [iv for iv in intervals if iv.R == N and iv.idx != i1.idx]
        if j_candidates:
            # minimal L among j_candidates
            LminN = min(iv.L for iv in j_candidates)
            if LminN <= i1.R + 1:
                # pick a j with L == LminN
                j = next(iv for iv in j_candidates if iv.L == LminN)
                ans[i1.idx - 1] = 1
                ans[j.idx - 1] = 1
                print(2)
                print(' '.join(str(x) for x in ans))
                return

    # 5) Fallback: try a 3-step solution via extremes and greedy cover of J2
    if M >= 2:
        i_star = max(intervals, key=lambda x: (x.L, -x.R)).idx
        j_star = min(intervals, key=lambda x: (x.R, -x.L)).idx
        L_star = max(iv.L for iv in intervals if iv.idx == i_star)
        R_star = min(iv.R for iv in intervals if iv.idx == j_star)
        JL = max(L_star, 1)
        JR = min(R_star, N)

        if JL <= JR:
            excluding = {i_star, j_star}
            pool = [(iv.L, iv.R, iv.idx) for iv in intervals if iv.idx not in excluding]
            pool.sort()

            import heapq
            current = JL
            p = 0
            heap = []
            used = []
            success = True
            while current <= JR:
                while p < len(pool) and pool[p][0] <= current:
                    Lp, Rp, idxp = pool[p]
                    heapq.heappush(heap, (-Rp, idxp, Lp, Rp))
                    p += 1
                found = False
                while heap:
                    negRp, idxp, Lp, Rp = heapq.heappop(heap)
                    if Rp <= current - 1:
                        continue
                    used.append(idxp)
                    current = Rp + 1
                    found = True
                    break
                if not found:
                    success = False
                    break
            if success:
                ans[i_star - 1] = 2
                ans[j_star - 1] = 2
                for u in used:
                    ans[u - 1] = 1
                K = 2 + len(used)
                print(K)
                print(' '.join(str(x) for x in ans))
                return

    print(-1)

if __name__ == "__main__":
    solve()
